# kidctl – feste Basisliste der Programm-Whitelist (für Eltern unsichtbar).
#
#   pkg:<paket>   Wurzelpaket: alle ausführbaren Dateien (/usr/bin, /usr/lib) seiner gesamten
#                 Abhängigkeitskette sind erlaubt – außer Programmen mit sichtbarem
#                 .desktop-Eintrag (Apps = Eltern-Entscheidung), Shells/Interpreter und Ausschlüssen.
#   group:<name>  alle installierten Pakete dieser Paketgruppe als Wurzelpakete (wie pkg:)
#   <muster>      zusätzlich erlaubter Pfad (fnmatch-Glob)
#   !<muster>     nie Basis (Eltern können es trotzdem einzeln freigeben)
#
# Dazu kommen automatisch Programme, die Pakete als Hintergrunddienst der Sitzung eintragen
# (/etc/xdg/autostart, D-Bus- und systemd-User-Dienste) – mit denselben Ausnahmen.
# Der Agent berechnet die Liste beim Start und nach jedem pacman-Update neu.
# /usr/lib/kidctl/* ist immer erlaubt, der dynamische Loader (ld-linux*) nie.

# --- Plasma-Sitzung (Wayland) ---
# Gruppen decken die Hintergrunddienste einer vollen Plasma-Installation ab (kded, kwalletd,
# powerdevil, baloo, kactivitymanagerd, Portale, KIO-Worker …); sichtbare Apps daraus (Discover,
# Systemmonitor …) bleiben Eltern-Entscheidung.
group:plasma
group:kf6
pkg:plasma-desktop
pkg:plasma-workspace
pkg:kwin
pkg:xorg-xwayland
pkg:plasma-nm
pkg:plasma-pa
pkg:kscreen
pkg:kde-gtk-config
pkg:breeze-gtk
# --- GNOME-Sitzung (Wayland) ---
# Gruppe deckt die Hintergrunddienste einer vollen GNOME-Installation ab (gsd-*, gnome-shell-Helfer,
# Tracker/LocalSearch, gvfs, Evolution-Data-Server …); sichtbare Apps daraus (Dateien, Software,
# Einstellungen …) bleiben Eltern-Entscheidung. Nicht installierte Pakete/Gruppen werden übersprungen.
group:gnome
pkg:gdm
pkg:gnome-shell
pkg:mutter
pkg:gnome-session
pkg:gnome-settings-daemon
pkg:xdg-desktop-portal-gnome
pkg:gnome-keyring
# Hilfsskripte (gjs) von GNOME Shell und systemweit installierten Erweiterungen (z. B. AppIndicator)
/usr/share/gnome-shell/*
# --- Audio, Portale ---
pkg:pipewire
pkg:pipewire-pulse
pkg:wireplumber
pkg:xdg-desktop-portal-kde
pkg:xdg-desktop-portal-gtk
# --- Flatpak-Laufzeit (Apps selbst über App-ID geregelt) ---
pkg:flatpak

# --- harmlose Hilfsprogramme, die Skripte (z. B. steam.sh) aufrufen ---
/usr/bin/which

# --- vom Agenten im Namen des Kindes gestartet (Benachrichtigung, Abmeldung über Plasma/GNOME) ---
/usr/bin/notify-send
/usr/bin/busctl

# --- xrdb (Start von Xwayland) ruft den C-Präprozessor auf, falls gcc installiert ist ---
/usr/bin/cpp
/usr/lib/gcc/*/*/cc1
