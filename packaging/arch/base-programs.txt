# kidctl – feste Basisliste der Programm-Whitelist (für Eltern unsichtbar).
#
#   pkg:<paket>   Wurzelpaket: alle ausführbaren Dateien (/usr/bin, /usr/lib) seiner gesamten
#                 Abhängigkeitskette sind erlaubt – außer Programmen mit sichtbarem
#                 .desktop-Eintrag (Apps = Eltern-Entscheidung), Shells/Interpreter und Ausschlüssen.
#   group:<name>  alle installierten Pakete dieser Paketgruppe als Wurzelpakete (wie pkg:)
#   <muster>      zusätzlich erlaubter Pfad (fnmatch-Glob)
#   !<muster>     nie Basis (Eltern können es trotzdem einzeln freigeben)
#
# Der Agent berechnet die Liste beim Start und nach jedem pacman-Update neu.
# /usr/lib/kidctl/* ist immer erlaubt, der dynamische Loader (ld-linux*) nie.

# --- Plasma-Sitzung (Wayland) ---
# Gruppen decken die Hintergrunddienste einer vollen Plasma-Installation ab (kded, kwalletd,
# powerdevil, baloo, kactivitymanagerd, Portale, KIO-Worker …); sichtbare Apps daraus (Discover,
# Systemmonitor …) bleiben Eltern-Entscheidung.
group:plasma
group:kf6
pkg:plasma-desktop
pkg:plasma-workspace
pkg:kwin
pkg:xorg-xwayland
pkg:plasma-nm
pkg:plasma-pa
pkg:kscreen
pkg:kde-gtk-config
pkg:breeze-gtk
# --- Audio, Portale ---
pkg:pipewire
pkg:pipewire-pulse
pkg:wireplumber
pkg:xdg-desktop-portal-kde
pkg:xdg-desktop-portal-gtk
# --- Flatpak-Laufzeit (Apps selbst über App-ID geregelt) ---
pkg:flatpak

# --- vom Agenten im Namen des Kindes gestartet (Benachrichtigung, Plasma-Abmeldung) ---
/usr/bin/notify-send
/usr/bin/busctl

# --- xrdb (Start von Xwayland) ruft den C-Präprozessor auf, falls gcc installiert ist ---
/usr/bin/cpp
/usr/lib/gcc/*/*/cc1

# --- Terminals nie als Basis ---
!/usr/bin/konsole
!/usr/bin/yakuake
!/usr/bin/xterm
!/usr/bin/alacritty
!/usr/bin/kitty
!/usr/bin/foot

# --- universelle Netzwerk-Werkzeuge (kommen über Gruppen-Abhängigkeiten mit) nie als Basis ---
