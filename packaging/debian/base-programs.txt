# kidctl – feste Basisliste der Programm-Whitelist für Debian/Ubuntu/Mint/Pop!_OS (für Eltern unsichtbar).
# Format wie packaging/arch/base-programs.txt. Paketgruppen gibt es hier nicht (nur die Pseudogruppe
# „essential“): Metapakete der Desktops ziehen den Rest über Depends und Recommends nach (nur installierte
# Pakete zählen).
#
#   pkg:<paket>   Wurzelpaket: alle ausführbaren Dateien seiner gesamten Abhängigkeitskette – außer
#                 Programmen mit sichtbarem .desktop-Eintrag (Apps = Eltern-Entscheidung),
#                 Shells/Interpreter und Ausschlüssen.
#   <muster>      zusätzlich erlaubter Pfad (fnmatch-Glob)
#   !<muster>     nie Basis (Eltern können es trotzdem einzeln freigeben)

# --- Grundsystem: Essential-Pakete (grep, sed, coreutils, …) – hängen von nichts ab, sind aber immer da ---
group:essential
